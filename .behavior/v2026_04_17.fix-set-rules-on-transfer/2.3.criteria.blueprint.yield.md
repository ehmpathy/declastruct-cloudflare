- usecase.2 = custom redirect rules ✓

- usecase.4 = error surface ✓

## subcomponent contracts

```
given('DeclaredCloudflareDomainRuleRedirect domain object')
  then('extends DomainEntity')
  then('has static primary = ["id"]')
  then('has static unique = ["zone", "slug"]')
  then('has static metadata = ["id"]')
  then('has static readonly = ["createdOn", "modifiedOn"]')
  then('has nested spec of type DeclaredCloudflareDomainRuleRedirectSpec')

given('DeclaredCloudflareDomainRuleRedirectSpec domain object')
  then('extends DomainLiteral')
  then('has expression: string')
  then('has parameters: PickOne<{ fromValue, fromList }>')
  then('fromValue.targetUrl is { value: string } | { expression: string }')
  then('fromValue.queryString is "preserve" | "ignore" | undefined')
  then('fromValue.statusCode is 301 | 302 | 303 | 307 | 308 | undefined')

given('DeclaredCloudflareDomainRuleRedirectDao')
  then('exposes: get.one.byUnique(input, context) => HasMetadata<Rule> | null')
  then('exposes: set.findsert(input, context) => HasMetadata<Rule>')
  then('exposes: set.upsert(input, context) => HasMetadata<Rule>')
  then('exposes: set.delete(input, context) => void')
  then('integrates with declastruct provider pattern')

given('getOneDomainRuleRedirect operation')
  then('exposes: getOneDomainRuleRedirect({ by: { unique } }, context) => Rule | null')
  then('looks up rule by zone + slug via cloudflare rulesets API')

given('setDomainRuleRedirect operation')
  then('exposes: setDomainRuleRedirect({ findsert | upsert }, context) => Rule')
  then('creates or updates rule in zone ruleset')
  then('transforms slug to cloudflare description field')
  then('transforms queryString to preserve_query_string boolean')

given('delDomainRuleRedirect operation')
  then('exposes: delDomainRuleRedirect({ by: { primary | unique } }, context) => void')
  then('removes rule from zone ruleset')

given('preset specs')
  then('exports RULE_REDIRECT_SPEC_HTTP_TO_HTTPS')
  then('exports RULE_REDIRECT_SPEC_ROOT_TO_WWW')
  then('both use queryString: "preserve" and statusCode: 301')
```
