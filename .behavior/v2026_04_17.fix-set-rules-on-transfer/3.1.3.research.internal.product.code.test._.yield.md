## pattern.3: getSampleCloudflareApiContext utility [REUSE]

### source

```typescript
// src/.test/getSampleCloudflareApiContext.ts:15-45
export const getSampleCloudflareApiContext = (): ContextCloudflareApi &
  ContextLogTrail => {
  const apiToken = process.env.CLOUDFLARE_API_TOKEN;
  const accountId = process.env.CLOUDFLARE_ACCOUNT_ID;

  if (!apiToken)
    BadRequestError.throw('CLOUDFLARE_API_TOKEN env var required...');
  if (!accountId)
    BadRequestError.throw('CLOUDFLARE_ACCOUNT_ID env var required...');

  const client = new Cloudflare({ apiToken });

  return {
    cloudflare: { client, accountId },
    log: { debug: () => {}, info: () => {}, warn: () => {}, error: () => {} },
  };
};
```

### relevance

redirect rule integration tests will use same context factory. no modification needed.

## pattern.4: integration tests with given/when/then [REUSE]

### source

```typescript
// src/domain.operations/domainDnsRecord/domainDnsRecord.integration.test.ts:17-92
describe('domainDnsRecord', () => {
  const hasCredentials =
    process.env.CLOUDFLARE_API_TOKEN && process.env.CLOUDFLARE_ACCOUNT_ID;
  const describeWithCredentials = hasCredentials ? describe : describe.skip;

  describeWithCredentials('with live cloudflare api', () => {
    const getContext = () => getSampleCloudflareApiContext();

    const zonesData = useBeforeAll(async () => {
      const allZones = await getAllDomainZones(getContext());
      return { zones: allZones, firstZone: allZones[0] ?? null };
    });

    given('a zone with DNS records', () => {
      when('[t0] getAllDomainDnsRecords is called by zone id', () => {
        then('it should return an array of records', async () => {
          if (!zonesData.firstZone) return;
          const records = await getAllDomainDnsRecords(
            { zone: { id: zonesData.firstZone.id } },
            getContext(),
          );
          expect(Array.isArray(records)).toBe(true);
        });
      });
    });
  });
});
```

### relevance

redirect rule integration tests follow same pattern:
- conditional skip via `hasCredentials`
- `useBeforeAll` for setup data
- given/when/then blocks
- skip if no zones via early return

## pattern.5: acceptance tests via CLI [REUSE]

### source

```typescript
// src/contract/sdks/declastruct.acceptance.test.ts:17-161
describe('declastruct CLI workflow', () => {
  if (!process.env.CLOUDFLARE_API_TOKEN)
    BadRequestError.throw('CLOUDFLARE_API_TOKEN not set...');

  const testDir = join(__dirname, '.test', '.temp', 'acceptance', ...);
  const resourcesFile = join(__dirname, '.test', 'assets', 'resources.acceptance.ts');
  const planFile = join(testDir, 'plan.json');

  beforeAll(() => mkdirSync(testDir, { recursive: true }));

  given('a declastruct resources file with cloudflare provider', () => {
    when('[t0] plan is run via declastruct CLI', () => {
      const prep = useBeforeAll(async () => {
        execSync(`npx declastruct plan --wish ${resourcesFile} --into ${planFile}`, ...);
        return { plan: JSON.parse(readFileSync(planFile, 'utf-8')) };
      });

      then('creates a valid plan file', () => {
        expect(existsSync(planFile)).toBe(true);
      });
    });
  });
});
```

### relevance

extend acceptance test resources file to include redirect rules. extant test structure handles new resources automatically via provider daos.
